public int binSearchByPhone(int phone) {
    Arrays.sort(cList);
    int low = 0, high = cList.length - 1, mid;
    while (high >= low) {
        mid = (low + high) / 2;
        if (phone == cList[mid].phone)
            return mid;
        else if (phone > cList[mid].phone)
            low = mid + 1;
        else
            high = mid - 1;
    }
    return -1;
}

// 索引结点类
static class BlockInfo {
    int blockBeginIndex;  // 块的起始下标
    int blockKey;         // 块关键字（前3位）
    int count;             // 块中元素数量
}

// 分块查找
public int blockSearchByPhone(int phone) {
    BlockInfo[] blocks = getIndexBlock();
    int key = Integer.parseInt(
        Integer.toString(phone).substring(0,3));
    int blockindex = -1;
    for (int i = 0; i < blocks.length; i++) {
        if (key == blocks[i].blockKey) {
            blockindex = i; break;
        }
    }
    if (blockindex != -1) {
        for (int i = blocks[blockindex].blockBeginIndex;
             i < blocks[blockindex].blockBeginIndex
                + blocks[blockindex].count; i++) {
            if (phone == cList[i].phone)
                return i;
        }
    }
    return -1;
}

// 哈希结点
private static class Node {
    Contacts data;
    Node next;
}

// 构建哈希表（链地址法）
public Node[] createHashTable() {
    int maxPrime = getMaxPrime();  // = 7
    Node[] hashtable = new Node[maxPrime];
    for (int i = 0; i < cList.length; i++) {
        Node node = new Node();
        node.data = cList[i];
        node.next = null;
        int hash = cList[i].phone % maxPrime;
        if (hashtable[hash] == null) {
            hashtable[hash] = node;
        } else {
            Node p = hashtable[hash];
            while (p.next != null) p = p.next;
            p.next = node;  // 尾插法
        }
    }
    return hashtable;
}

// 哈希查找
public Contacts HashSearch(int key) {
    Node[] hashtable = createHashTable();
    int hash = key % maxPrime;
    Node p = hashtable[hash];
    while (p != null && p.data.phone != key) {
        p = p.next;
    }
    return p != null ? p.data : null;
}
